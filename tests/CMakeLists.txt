add_executable(stitchkit_tests test_main.cpp)
target_include_directories(stitchkit_tests PRIVATE ${PROJECT_SOURCE_DIR}/src ${PROJECT_SOURCE_DIR}/third_party/vlfeat)
target_compile_definitions(stitchkit_tests PRIVATE VL_STATIC)
target_link_libraries(stitchkit_tests PRIVATE stitchkit::stitchkit Eigen3::Eigen)
add_executable(stitchkit_e2e e2e.cpp)
target_link_libraries(stitchkit_e2e PRIVATE stitchkit::stitchkit)
set(asset_dir "${CMAKE_CURRENT_SOURCE_DIR}/assets")
set(output_dir "${CMAKE_CURRENT_BINARY_DIR}/outputs")
foreach(name convolution types mesh solver registration rew geometric_energy raster io edges lineae_contract device_contract failures)
  add_test(NAME unit.${name} COMMAND stitchkit_tests ${name} "${asset_dir}" "${output_dir}/${name}")
  set_tests_properties(unit.${name} PROPERTIES LABELS "unit;cpu" TIMEOUT 60)
endforeach()
foreach(name cuda_parity lineae_model)
  add_test(NAME optional.${name} COMMAND stitchkit_tests ${name} "${asset_dir}" "${output_dir}/${name}")
  set_tests_properties(optional.${name} PROPERTIES SKIP_RETURN_CODE 77 LABELS "optional" TIMEOUT 120)
endforeach()
foreach(method rew niswgsp gesgsp)
  foreach(scenario translation chain parallax library town_hall ship)
    add_test(NAME e2e.${method}.${scenario} COMMAND stitchkit_e2e ${method} ${scenario} "${asset_dir}" "${output_dir}/e2e")
    set_tests_properties(e2e.${method}.${scenario} PROPERTIES LABELS "e2e;cpu" TIMEOUT 180)
  endforeach()
endforeach()
if(STITCHKIT_BUILD_CLI)
add_test(NAME e2e.cli COMMAND ${CMAKE_COMMAND} -DSTITCH=$<TARGET_FILE:stitch> -DASSETS=${asset_dir}
  -DOUT=${output_dir}/cli -P ${CMAKE_CURRENT_SOURCE_DIR}/cli_test.cmake)
set_tests_properties(e2e.cli PROPERTIES LABELS "e2e;cli" TIMEOUT 180)
endif()

add_test(NAME assets.integrity COMMAND ${CMAKE_COMMAND} -DASSETS=${asset_dir} -P ${CMAKE_CURRENT_SOURCE_DIR}/verify_assets.cmake)
set_tests_properties(assets.integrity PROPERTIES LABELS "assets" TIMEOUT 30)

  set_tests_properties(regression.${name} PROPERTIES LABELS "regression;cpu" TIMEOUT 120)
endforeach()
