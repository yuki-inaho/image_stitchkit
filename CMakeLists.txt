cmake_minimum_required(VERSION 3.24)
project(stitchkit VERSION 0.2.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
option(STITCHKIT_ENABLE_CUDA "Build the optional CUDA sampling/compositing backend" OFF)
option(STITCHKIT_ENABLE_LINEAE "Build the optional ONNX Runtime LINEAE detector" OFF)
option(STITCHKIT_ENABLE_SANITIZERS "Enable ASan and UBSan on Clang/GCC" OFF)
option(STITCHKIT_BUILD_PYTHON "Build the nanobind Python extension" OFF)
option(STITCHKIT_BUILD_CLI "Build the command-line application" ON)
option(STITCHKIT_INSTALL_CPP "Install the native SDK" ON)
option(STITCHKIT_BUILD_EXAMPLES "Build the library usage example" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 99)

find_package(Eigen3 3.3 CONFIG QUIET)
if(NOT TARGET Eigen3::Eigen)
  find_path(STITCHKIT_EIGEN_INCLUDE_DIR Eigen/Core HINTS ${Eigen3_INCLUDE_DIR} PATH_SUFFIXES eigen3 REQUIRED)
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${STITCHKIT_EIGEN_INCLUDE_DIR}")
endif()
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(Threads REQUIRED)

# Only the scalar SIFT subset is compiled; unused upstream platform/thread code is excluded.
add_library(stitchkit_vlfeat OBJECT third_party/vlfeat/vl/sift.c third_party/vlfeat/vl/imopv.c third_party/vlfeat/runtime.c)
target_compile_definitions(stitchkit_vlfeat PRIVATE VL_STATIC VL_DISABLE_SSE2 VL_DISABLE_AVX VL_DISABLE_OPENMP VL_DISABLE_THREADS)
target_include_directories(stitchkit_vlfeat PRIVATE third_party/vlfeat)
set_target_properties(stitchkit_vlfeat PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_definitions(stitchkit_vlfeat PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

add_library(stitchkit STATIC src/types.cpp src/io.cpp src/report.cpp src/jpeg_bridge.c src/features.cpp src/registration.cpp
  src/lineae.cpp src/solver.cpp src/mesh_warp.cpp src/rew.cpp src/raster.cpp src/structures.cpp src/stitcher.cpp
  $<TARGET_OBJECTS:stitchkit_vlfeat>)
add_library(stitchkit::stitchkit ALIAS stitchkit)
set_target_properties(stitchkit PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(stitchkit PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src third_party/vlfeat)
target_link_libraries(stitchkit PRIVATE $<BUILD_INTERFACE:Eigen3::Eigen> PNG::PNG JPEG::JPEG Threads::Threads)
target_compile_features(stitchkit PUBLIC cxx_std_17)
target_compile_definitions(stitchkit PRIVATE STITCHKIT_VERSION="${PROJECT_VERSION}")
target_compile_definitions(stitchkit PRIVATE VL_STATIC VL_DISABLE_SSE2 VL_DISABLE_AVX VL_DISABLE_OPENMP VL_DISABLE_THREADS)
if(UNIX)
  target_link_libraries(stitchkit PRIVATE m)
endif()
if(MSVC)
  target_compile_options(stitchkit PRIVATE /W4 /utf-8)
  target_compile_definitions(stitchkit PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  target_compile_options(stitchkit PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:-Wall;-Wextra;-Wpedantic>)
endif()

if(STITCHKIT_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(stitchkit PRIVATE src/render_cuda.cu)
  target_compile_definitions(stitchkit PRIVATE STITCHKIT_WITH_CUDA=1)
  target_link_libraries(stitchkit PRIVATE CUDA::cudart)
  set_target_properties(stitchkit PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
if(STITCHKIT_ENABLE_LINEAE)
  find_package(onnxruntime CONFIG REQUIRED)
  target_sources(stitchkit PRIVATE src/lineae_onnx.cpp)
  target_compile_definitions(stitchkit PRIVATE STITCHKIT_WITH_LINEAE=1)
  if(TARGET onnxruntime::onnxruntime)
    target_link_libraries(stitchkit PRIVATE onnxruntime::onnxruntime)
  elseif(TARGET onnxruntime)
    target_link_libraries(stitchkit PRIVATE onnxruntime)
  else()
    message(FATAL_ERROR "The ONNX Runtime package did not define a supported CMake target")
  endif()
endif()

if(STITCHKIT_ENABLE_SANITIZERS)
  if(MSVC OR STITCHKIT_ENABLE_CUDA)
    message(FATAL_ERROR "This sanitizer preset supports CPU Clang/GCC builds only")
  endif()
  foreach(t stitchkit stitchkit_vlfeat)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  target_compile_options(stitchkit INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(stitchkit PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

if(STITCHKIT_BUILD_CLI)
  add_executable(stitch app/main.cpp)
  target_link_libraries(stitch PRIVATE stitchkit::stitchkit)
endif()
if(STITCHKIT_BUILD_EXAMPLES)
  add_executable(stitchkit_example examples/basic.cpp)
  target_link_libraries(stitchkit_example PRIVATE stitchkit::stitchkit)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()

if(STITCHKIT_INSTALL_CPP)
set(native_install_targets stitchkit)
if(STITCHKIT_BUILD_CLI)
  list(APPEND native_install_targets stitch)
endif()
install(TARGETS ${native_install_targets} EXPORT stitchkitTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT stitchkitTargets NAMESPACE stitchkit:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stitchkit)
configure_package_config_file(cmake/stitchkitConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/stitchkitConfig.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stitchkit)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/stitchkitConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/stitchkitConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/stitchkitConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stitchkit)
install(FILES third_party/vlfeat/COPYING DESTINATION ${CMAKE_INSTALL_DATADIR}/stitchkit/licenses RENAME VLFeat-BSD.txt)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/stitchkit/licenses)

endif() # STITCHKIT_INSTALL_CPP

if(STITCHKIT_BUILD_PYTHON)
  find_package(Python 3.10 COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
    OUTPUT_VARIABLE nanobind_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE
    COMMAND_ERROR_IS_FATAL ANY)
  find_package(nanobind 2.15 CONFIG REQUIRED)
  nanobind_add_module(_native NB_STATIC bindings/module.cpp)
  target_link_libraries(_native PRIVATE stitchkit::stitchkit)
  target_compile_definitions(_native PRIVATE STITCHKIT_VERSION="${PROJECT_VERSION}"
    STITCHKIT_COMPILER_INFO="${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}"
    STITCHKIT_BUILD_CONFIGURATION="$<CONFIG>")
  install(TARGETS _native LIBRARY DESTINATION stitchkit RUNTIME DESTINATION stitchkit COMPONENT python)
endif()

if(STITCHKIT_BUILD_PYTHON)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION stitchkit/licenses COMPONENT python)
  install(FILES third_party/vlfeat/COPYING DESTINATION stitchkit/licenses RENAME VLFeat-BSD.txt COMPONENT python)
endif()
